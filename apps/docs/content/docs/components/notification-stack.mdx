## Usage

```tsx
import {
  NotificationStack,
  type NotificationData
} from '@/components/matos-ui/notification-stack'
```

```tsx
const [notifications, setNotifications] = useState<NotificationData[]>([])

function dismiss(id: string) {
  setNotifications(prev => prev.filter(n => n.id !== id))
}

;<NotificationStack
  notifications={notifications}
  onDismiss={dismiss}
  size="md"
/>
```

### With Provider

To trigger notifications from anywhere in the app:

```tsx
import {
  NotificationStackProvider,
  useNotificationStack
} from '@/components/matos-ui/notification-stack'
```

```tsx
<NotificationStackProvider>
  <App />
</NotificationStackProvider>
```

```tsx
function MyComponent() {
  const { add, dismiss, dismissAll } = useNotificationStack()

  function handleClick() {
    add({
      app: 'Vercel',
      title: 'Deploy complete',
      description: 'Project deployed successfully.',
      timestamp: 'just now'
    })
  }

  return <button onClick={handleClick}>Notify</button>
}
```

## Reference

### NotificationStack Props

| Prop             | Type                   | Default | Description                         |
| ---------------- | ---------------------- | ------- | ----------------------------------- |
| `notifications`  | `NotificationData[]`   | —       | List of notifications.              |
| `onDismiss`      | `(id: string) => void` | —       | Callback when dismissed.            |
| `size`           | `"sm" \| "md" \| "lg"` | `"md"`  | Stack width.                        |
| `maxStackLayers` | `number`               | `3`     | Visible layers behind the top card. |

### NotificationData

| Field         | Type                  | Description                              |
| ------------- | --------------------- | ---------------------------------------- |
| `id`          | `string`              | Unique identifier.                       |
| `app`         | `string?`             | App name (displayed as uppercase label). |
| `title`       | `string`              | Title / sender name.                     |
| `description` | `string?`             | Notification message.                    |
| `avatar`      | `ReactNode?`          | Avatar or icon on the left.              |
| `timestamp`   | `string?`             | Time text (e.g., "2 min ago").           |
| `action`      | `{ label, onClick }?` | Action button inside the notification.   |

### Interactions

- **Drag** the card up, down, or sideways to dismiss
- **Click ×** to dismiss immediately
- Background layers automatically re-stack with a spring animation
- Cards follow an inset layout: loose header, recessed content panel, and loose footer

Also exported: `notificationStackVariants`, `notificationCardVariants` and all types.
